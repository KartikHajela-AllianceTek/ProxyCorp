class DualTranscriberAgent:
    def __init__(self):
        self.deepgram = DeepgramClient(os.getenv("DEEPGRAM_API_KEY"))
        self.dg_connection = None 
        
        # --- THE PROXYCORP STATE MACHINE ---
        self.current_engine = "sarvam_hi" # START WITH SARVAM BY DEFAULT
        self.english_start_time = None    # Tracks sustained English
        
        self.sarvam_buffer = bytearray()
        self.is_listening = False
        self.last_spoken_time = 0
        self.cooldown_seconds = 4 # Slightly longer to capture full hybrid sentences
        self.noise_threshold_rms = 500  
        self._loop = None
        self.on_transcript_callback = None 

    def _calculate_rms(self, pcm_bytes: bytes) -> float:
        count = len(pcm_bytes) // 2
        if count == 0: return 0.0
        try:
            shorts = struct.unpack(f'<{count}h', pcm_bytes)
            return math.sqrt(sum(s * s for s in shorts) / count)
        except:
            return 0.0

    async def run(self):
        self._loop = asyncio.get_running_loop()
        logger.info("🛡️ Dual-Engine Gatekeeper active. Defaulting to Sarvam Hybrid.")
        while True:
            await asyncio.sleep(0.5)
            # Only flush Sarvam if it's the active engine (saves credits)
            if self.current_engine.startswith("sarvam") and len(self.sarvam_buffer) > 0:
                if time.time() - self.last_spoken_time > self.cooldown_seconds:
                    asyncio.create_task(self._flush_sarvam_buffer())

    async def _flush_sarvam_buffer(self):
        logger.info(f"📦 Flushing {len(self.sarvam_buffer)} bytes to Sarvam AI...")
        lang_code = "hi-IN" if "hi" in self.current_engine else "gu-IN"
        
        buffer_copy = bytes(self.sarvam_buffer)
        self.sarvam_buffer.clear()
        self.is_listening = False
        
        transcript = await asyncio.to_thread(transcribe_with_sarvam, buffer_copy, lang_code)
        
        if transcript:
            logger.info(f"🗣️ [SARVAM HYBRID]: {transcript}")
            if self.on_transcript_callback:
                 await self.on_transcript_callback(transcript)

    def _start_deepgram(self):
        if self.dg_connection: return
        logger.info("🔊 Waking up Deepgram...")
        self.dg_connection = self.deepgram.listen.live.v("1")

        def on_message(self_obj, result, **kwargs):
            sentence = result.channel.alternatives[0].transcript if result.channel.alternatives else ""
            if sentence.strip():
                logger.info(f"🗣️ [DEEPGRAM EN]: {sentence}")
                if self.on_transcript_callback and self._loop:
                    asyncio.run_coroutine_threadsafe(self.on_transcript_callback(sentence), self._loop)

        self.dg_connection.on("Results", on_message)
        options = LiveOptions(**get_deepgram_config())
        if self.dg_connection.start(options):
            pass # Listening state handled by VAD

    def _stop_deepgram(self):
        if not self.dg_connection: return
        self.dg_connection.finish()
        self.dg_connection = None

    async def send_audio(self, pcm_bytes: bytes):
        current_time = time.time()
        rms_volume = self._calculate_rms(pcm_bytes)

        if rms_volume > self.noise_threshold_rms:
            self.last_spoken_time = current_time
            self.is_listening = True

            # --- ROUTING LOGIC ---
            if self.current_engine == "deepgram":
                # Pure English Mode: Sarvam is sleeping
                if not self.dg_connection: self._start_deepgram()
                self.dg_connection.send(pcm_bytes)
            
            else:
                # Hybrid Mode: Feed Sarvam
                self.sarvam_buffer.extend(pcm_bytes)
                
                # FADE IN: If English has been spoken for > 10s, start warming up Deepgram
                if self.english_start_time and (current_time - self.english_start_time > 10):
                    if not self.dg_connection: self._start_deepgram()
                    self.dg_connection.send(pcm_bytes) # Both receive audio temporarily

        # Deepgram Cooldown Check
        if self.dg_connection and (current_time - self.last_spoken_time > self.cooldown_seconds):
            self._stop_deepgram()

        parts = [Part(root=TextPart(text=status_msg))]

import os
import requests
import io
import wave
import logging
from typing import Dict, Any

logger = logging.getLogger(__name__)

def get_deepgram_config() -> Dict[str, Any]:
    """Configures Deepgram Nova-3 for English."""
    return {
        "model": "nova-3",
        "language": "en-US",
        "smart_format": True,
        "diarize": True, 
        "interim_results": True,
        "encoding": "linear16",
        "sample_rate": 16000,
        "channels": 1
    }

def create_wav_buffer(pcm_bytes: bytes, sample_rate: int = 16000) -> io.BytesIO:
    """Wraps raw PCM bytes into a valid WAV file in memory for Sarvam AI."""
    wav_io = io.BytesIO()
    with wave.open(wav_io, 'wb') as wav_file:
        wav_file.setnchannels(1)
        wav_file.setsampwidth(2) # 16-bit
        wav_file.setframerate(sample_rate)
        wav_file.writeframes(pcm_bytes)
    wav_io.seek(0)
    return wav_io

def transcribe_with_sarvam(pcm_bytes: bytes, language_code: str) -> str:
    """
    Sends buffered audio to Sarvam's speech-to-text API.
    language_code: 'hi-IN' for Hindi, 'gu-IN' for Gujarati.
    """
    sarvam_api_key = os.getenv("SARVAM_API_KEY")
    if not sarvam_api_key:
        logger.error("Missing SARVAM_API_KEY")
        return ""

    # Sarvam's STT endpoint (refer to their latest docs for exact URL)
    url = "https://api.sarvam.ai/speech-to-text-translate" 
    
    wav_buffer = create_wav_buffer(pcm_bytes)
    
    headers = {"api-subscription-key": sarvam_api_key}
    files = {'file': ('audio.wav', wav_buffer, 'audio/wav')}
    data = {'prompt': 'Transcribe accurately in Latin script if possible.'}
    
    try:
        response = requests.post(url, headers=headers, files=files, data=data)
        if response.status_code == 200:
            return response.json().get("transcript", "")
        else:
            logger.error(f"Sarvam API Error: {response.text}")
            return ""
    except Exception as e:
        logger.error(f"Failed to reach Sarvam: {e}")
        return ""
